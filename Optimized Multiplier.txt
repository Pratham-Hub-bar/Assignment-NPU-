module optimized_multiplier(
    input [3:0] m_plicand, m_plier,
    output reg [7:0] product);

    reg [7:0] temp;			// * bit register to handle the shift operations
    
    integer i;
    

    always @(*) begin			
        product = 0; // Initially the product is zero.

        temp [7:0] = {4'b0000, m_plicand};   //Using a 8 bit register to store the value of multiplicand and so that it can handle shift operation too
 
        for(i=0; i<4; i=i+1) begin: Control_Unit //Traveling through the multiplier bits from 0 to the final bit
            if(m_plier[i] == 1)  begin		// If the multiplier bit is 1 then add the temp to partial product, else no need to add it 
                product = product + temp;
            end

            temp[7:0] = {temp[6:0], 1'b0};	//Increment the multiplicand no matter what
        end
    end
endmodule
