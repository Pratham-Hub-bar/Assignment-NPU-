module round_robin_arbiter_tb();
    wire [3:0] gnt;
    reg [3:0] req;
    reg clk, reset;
    integer i;
    round_robin_arbiter uut(.gnt(gnt), .req(req), .clk(clk), .reset(reset));

    always begin #10 clk = ~clk; end

    initial begin
        clk=0;
        req = 4'b1111;
        reset = 1;
        #20;
        reset = 0;
        for(i = 0; i<16; i=i+1) begin
            #20 req = i;
        end

        $stop;
    end
endmodule
